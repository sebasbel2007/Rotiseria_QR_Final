rules_version = '2';


service cloud.firestore {


  match /databases/{database}/documents {





    function getRole() {


      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.rol;
    }

    match /users/{userId} {
      allow read: if request.auth != null &&
        (request.auth.uid == userId || getRole() == 'admin');
      allow create: if request.auth != null && request.auth.uid == userId;
      allow update, delete: if request.auth != null && getRole() == 'admin';
    }

    match /productos/{productoId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && getRole() == 'admin';
    }

    match /pedidos/{pedidoId} {
      allow create: if request.auth != null && (
        request.resource.data.cliente_uid == request.auth.uid ||
        getRole() in ['cajero', 'admin']
      );
      allow read: if request.auth != null && (
        resource.data.cliente_uid == request.auth.uid ||
        getRole() in ['cocina', 'cajero', 'admin']
      );
      allow update: if request.auth != null && (
        getRole() == 'admin' ||
        getRole() == 'cajero' ||
        (getRole() == 'cocina' &&
          request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['estado_cocina', 'motivo_cancelacion']))
      );
      allow delete: if request.auth != null && getRole() == 'admin';
    }

    match /config/{docId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && getRole() == 'admin';
    }
  }
}
