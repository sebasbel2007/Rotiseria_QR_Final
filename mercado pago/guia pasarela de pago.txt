Backend (Node.js + Express)
server.js — El servidor que habla con MercadoPago:

// Inicialización del cliente MP
const mpClient = new MercadoPagoConfig({
  accessToken: process.env.MP_ACCESS_TOKEN
});
const preferenceClient = new Preference(mpClient);
const paymentClient    = new Payment(mpClient);

// Endpoint: Crear preferencia de pago
app.post('/api/create-preference', async (req, res) => {
  const { type, userId, courseTitle, price, courseId, studentId, teacherId } = req.body;
  
  const BASE_URL = `${process.env.APP_URL}/pages/checkout/checkout-result.html`;
  
  // Construir body según tipo de pago
  if (type === 'membership') {
    body = {
      items: [{ title: 'Membresía Profesor', quantity: 1, unit_price: 20000, currency_id: 'ARS' }],
      back_urls: {
        success: `${BASE_URL}?type=membership&userId=${userId}&status=approved`,
        failure: `${BASE_URL}?type=membership&status=failure`,
        pending: `${BASE_URL}?type=membership&status=pending`
      },
      auto_return: 'approved',
      metadata: { type: 'membership', userId }
    };
  } else if (type === 'course') {
    body = {
      items: [{ title: courseTitle, quantity: 1, unit_price: price, currency_id: 'ARS' }],
      back_urls: {
        success: `${BASE_URL}?type=course&courseId=${courseId}&studentId=${studentId}&teacherId=${teacherId}&price=${price}&status=approved`,
        failure: `${BASE_URL}?type=course&courseId=${courseId}&status=failure`,
        pending: `${BASE_URL}?type=course&courseId=${courseId}&status=pending`
      },
      auto_return: 'approved',
      metadata: { type: 'course', courseId, studentId, teacherId, price }
    };
  }
  
  const preference = await preferenceClient.create({ body });
  res.json({ id: preference.id, init_point: preference.init_point, sandbox_init_point: preference.sandbox_init_point });
});

// Endpoint: Webhook de MP
app.post('/api/webhook', async (req, res) => {
  // Validar firma HMAC-SHA256
  const xSignature = req.headers['x-signature'];
  const xRequestId = req.headers['x-request-id'];
  
  if (xSignature && xRequestId && process.env.MP_WEBHOOK_SECRET) {
    const [tsPart, v1Part] = xSignature.split(',');
    const ts = tsPart?.split('=')[1];
    const v1 = v1Part?.split('=')[1];
    const manifest = `id:${data?.id};request-id:${xRequestId};ts:${ts};`;
    const expected = crypto.createHmac('sha256', process.env.MP_WEBHOOK_SECRET)
      .update(manifest).digest('hex');
    if (expected !== v1) return res.sendStatus(401);
  }
  
  res.sendStatus(200); // Responder rápido a MP
  
  if (type === 'payment' && data?.id) {
    const payment = await paymentClient.get({ id: data.id });
    console.log(`Pago ${payment.id} — status: ${payment.status}`);
  }
});
Frontend (HTML + JS)
payments.js — Lógica de pago del lado del cliente:

// Constantes de comisión
const PLATFORM_COMMISSION = 0.30;
const TEACHER_SHARE       = 0.70;
const MEMBERSHIP_PRICE    = 20000; // ARS

// Crear preferencia via servidor
export async function createPreference(type, data) {
  const res = await fetch('/api/create-preference', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ type, ...data })
  });
  if (!res.ok) throw new Error('No se pudo crear la preferencia de pago.');
  return res.json();
}

// Procesar pago aprobado de membresía
export async function processMembershipPayment(userId, paymentId) {
  await upgradeToTeacher(userId);
  await createTransaction({
    type: 'membership', studentId: userId, teacherId: null, courseId: null,
    amount: MEMBERSHIP_PRICE, platformAmount: MEMBERSHIP_PRICE, teacherAmount: 0,
    paymentId, status: 'completed'
  });
  await createNotification(userId, 'enrollment', '¡Bienvenido, Profesor!', 'Tu membresía fue activada.');
}

// Procesar pago aprobado de curso
export async function processCoursePayment(studentId, courseId, teacherId, price, paymentId, totalModules) {
  const platformAmount = Math.round(price * PLATFORM_COMMISSION);
  const teacherAmount  = Math.round(price * TEACHER_SHARE);
  
  await createEnrollment(studentId, courseId, teacherId, price, paymentId, totalModules);
  await createTransaction({
    type: 'course_purchase', studentId, teacherId, courseId,
    amount: price, platformAmount, teacherAmount, paymentId, status: 'completed'
  });
  await createNotification(studentId, 'enrollment', '¡Compra exitosa!', 'Ya podés acceder al curso.');
  await createNotification(teacherId, 'sale', 'Nueva venta', `Ganaste $${teacherAmount} ARS.`);
}

// Renderizar botón de Checkout Pro
export async function renderCheckoutButton(containerId, preferenceId, initPoint) {
  const container = document.getElementById(containerId);
  container.innerHTML = `
    <a href="${initPoint}" style="...estilos...">
      Pagar con Mercado Pago
    </a>`;
}

Base de Datos (Firestore)
Colección transactions:

{
  type: 'course_purchase' | 'membership' | 'refund',
  studentId: 'uid_del_estudiante',
  teacherId: 'uid_del_profesor' | null,
  courseId: 'id_del_curso' | null,
  amount: 10000,           // Monto total en ARS
  platformAmount: 3000,    // 30% para la plataforma
  teacherAmount: 7000,     // 70% para el profesor
  paymentId: 'id_de_mp',   // ID del pago en MercadoPago

  updatedAt: Timestamp
